## 資料庫設計

### 資料表結構

#### 1. users（使用者表）
```sql
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  name VARCHAR(100),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2. body_compositions（身體組成表）
```sql
CREATE TABLE body_compositions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  height DECIMAL(5,2) NOT NULL,           -- 身高 (cm)
  weight DECIMAL(5,2) NOT NULL,           -- 體重 (kg)
  age INTEGER NOT NULL,
  gender VARCHAR(10) NOT NULL,            -- 'male' | 'female'
  activity_level VARCHAR(20) NOT NULL,    -- 'sedentary' | 'light' | 'moderate' | 'active' | 'veryActive'
  body_fat DECIMAL(5,2),                 -- 體脂率 (%)
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 3. goals（目標設定表）
```sql
CREATE TABLE goals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  goal_type VARCHAR(20) NOT NULL,        -- 'lose' | 'gain' | 'maintain'
  target_weight DECIMAL(5,2),
  target_date DATE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 4. nutrition_requirements（營養需求表）
```sql
CREATE TABLE nutrition_requirements (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  daily_calories DECIMAL(8,2) NOT NULL,
  protein DECIMAL(8,2) NOT NULL,         -- 蛋白質 (g)
  carbohydrates DECIMAL(8,2) NOT NULL,   -- 碳水化合物 (g)
  fat DECIMAL(8,2) NOT NULL,             -- 脂肪 (g)
  calculated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 5. foods（食物資料庫表）
```sql
CREATE TABLE foods (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  brand VARCHAR(100),
  calories_per_100g DECIMAL(8,2) NOT NULL,
  protein_per_100g DECIMAL(8,2) NOT NULL,
  carbohydrates_per_100g DECIMAL(8,2) NOT NULL,
  fat_per_100g DECIMAL(8,2) NOT NULL,
  fiber DECIMAL(8,2),                    -- 纖維 (g)
  sugar DECIMAL(8,2),                    -- 糖 (g)
  category VARCHAR(50),                  -- 食物分類
  is_custom BOOLEAN DEFAULT FALSE,      -- 是否為使用者自訂
  created_by UUID REFERENCES users(id),   -- 建立者（自訂食物）
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 6. food_logs（飲食記錄表）
```sql
CREATE TABLE food_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  food_id UUID REFERENCES foods(id) ON DELETE CASCADE,
  date DATE NOT NULL,                    -- 記錄日期
  meal_type VARCHAR(20) NOT NULL,        -- 'breakfast' | 'lunch' | 'dinner' | 'snack'
  quantity DECIMAL(8,2) NOT NULL,        -- 份量/重量 (g)
  calories DECIMAL(8,2) NOT NULL,
  protein DECIMAL(8,2) NOT NULL,
  carbohydrates DECIMAL(8,2) NOT NULL,
  fat DECIMAL(8,2) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 索引優化
CREATE INDEX idx_food_logs_user_date ON food_logs(user_id, date);
CREATE INDEX idx_food_logs_date ON food_logs(date);
```

#### 7. meals（餐點資料庫表）
```sql
CREATE TABLE meals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  description TEXT,
  meal_type VARCHAR(20) NOT NULL,        -- 'breakfast' | 'lunch' | 'dinner' | 'snack'
  calories DECIMAL(8,2) NOT NULL,
  protein DECIMAL(8,2) NOT NULL,
  carbohydrates DECIMAL(8,2) NOT NULL,
  fat DECIMAL(8,2) NOT NULL,
  image_url VARCHAR(500),                 -- 餐點圖片
  category VARCHAR(50),                   -- 餐點分類（低卡、高蛋白等）
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 8. meal_foods（餐點食物關聯表）
```sql
CREATE TABLE meal_foods (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  meal_id UUID REFERENCES meals(id) ON DELETE CASCADE,
  food_id UUID REFERENCES foods(id) ON DELETE CASCADE,
  quantity DECIMAL(8,2) NOT NULL,         -- 該食物在餐點中的份量
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 9. ai_chat_sessions（AI 聊天會話表）
```sql
CREATE TABLE ai_chat_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 10. ai_chat_messages（AI 聊天訊息表）
```sql
CREATE TABLE ai_chat_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES ai_chat_sessions(id) ON DELETE CASCADE,
  role VARCHAR(20) NOT NULL,  -- 'user' | 'assistant' | 'system'
  content TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 11. ai_analyses（AI 分析記錄表）
```sql
CREATE TABLE ai_analyses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  date DATE NOT NULL,
  analysis_type VARCHAR(50) NOT NULL,  -- 'nutrition_status' | 'meal_recommendation'
  input_data JSONB,
  ai_response JSONB,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

### AI 功能實作架構

#### 1. AI 聊天服務

```typescript
// backend/src/services/aiChatService.ts
import OpenAI from 'openai';

export class AIChatService {
  private openai: OpenAI;
  
  constructor() {
    this.openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY,
    });
  }

  async chatWithNutritionist(
    userId: string,
    message: string,
    conversationHistory: Message[]
  ): Promise<string> {
    const systemPrompt = `你是一位專業的營養師助手。根據使用者的飲食記錄和目標，提供專業建議。`;
    
    const response = await this.openai.chat.completions.create({
      model: "gpt-4",
      messages: [
        { role: "system", content: systemPrompt },
        ...conversationHistory,
        { role: "user", content: message }
      ],
      temperature: 0.7,
    });

    return response.choices[0].message.content;
  }
}
```

#### 2. AI 營養分析服務

```typescript
// backend/src/services/aiAnalysisService.ts
import OpenAI from 'openai';

export class AIAnalysisService {
  private openai: OpenAI;

  constructor() {
    this.openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY,
    });
  }

  async analyzeNutritionStatus(
    userId: string,
    dailyNutrition: NutritionData
  ): Promise<NutritionAnalysis> {
    const prompt = `
      分析使用者的營養攝取狀況：
      熱量：${dailyNutrition.calories} / ${dailyNutrition.targetCalories} kcal
      蛋白質：${dailyNutrition.protein} / ${dailyNutrition.targetProtein} g
      碳水化合物：${dailyNutrition.carbs} / ${dailyNutrition.targetCarbs} g
      脂肪：${dailyNutrition.fat} / ${dailyNutrition.targetFat} g
      
      請判斷：
      1. 哪些營養素過量
      2. 哪些營養素不足
      3. 提供改善建議
    `;

    const response = await this.openai.chat.completions.create({
      model: "gpt-4",
      messages: [
        { role: "system", content: "你是營養分析專家..." },
        { role: "user", content: prompt }
      ],
      response_format: { type: "json_object" },
    });

    return JSON.parse(response.choices[0].message.content);
  }
}
```

#### 3. AI 餐點推薦服務

```typescript
// backend/src/services/aiRecommendationService.ts
import OpenAI from 'openai';

export class AIRecommendationService {
  private openai: OpenAI;

  constructor() {
    this.openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY,
    });
  }

  async recommendMeals(
    userId: string,
    remainingCalories: number,
    remainingNutrition: NutritionData,
    mealType: 'breakfast' | 'lunch' | 'dinner' | 'snack'
  ): Promise<MealRecommendation[]> {
    const prompt = `
      根據以下條件推薦餐點：
      - 剩餘熱量：${remainingCalories} kcal
      - 剩餘蛋白質：${remainingNutrition.protein} g
      - 剩餘碳水化合物：${remainingNutrition.carbs} g
      - 剩餘脂肪：${remainingNutrition.fat} g
      - 用餐時段：${mealType}
      
      請推薦 3-5 個符合需求的餐點，並說明推薦理由。
    `;

    const response = await this.openai.chat.completions.create({
      model: "gpt-4",
      messages: [
        { role: "system", content: "你是餐點推薦專家..." },
        { role: "user", content: prompt }
      ],
      response_format: { type: "json_object" },
    });

    return JSON.parse(response.choices[0].message.content).recommendations;
  }
}
```

      name: "使用者名稱"

  }
}

// 範例：建立資源
{

#### 實作範例

```typescript
// backend/src/utils/response.ts

export interface ApiResponse<T> {
  success: boolean;
  data?: T;
  error?: {
    code: string;
    message: string;
    details?: any;
  };
  message?: string;
}

export interface PaginatedResponse<T> {
  success: boolean;
  data: {
    items: T[];
    pagination: {
      page: number;
      limit: number;
      total: number;
      totalPages: number;
    };
  };
}

// 成功回應輔助函式
export function successResponse<T>(
  data: T,
  message?: string
): ApiResponse<T> {
  return {
    success: true,
    data,
    message,
  };
}

// 錯誤回應輔助函式
export function errorResponse(
  code: string,
  message: string,
  details?: any
): ApiResponse<null> {
  return {
    success: false,
    error: {
      code,
      message,
      details,
    },
  };
}

// 分頁回應輔助函式
export function paginatedResponse<T>(
  items: T[],
  page: number,
  limit: number,
  total: number
): PaginatedResponse<T> {
  return {
    success: true,
    data: {
      items,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    },
  };
}
```

#### 使用範例

```typescript
// backend/src/controllers/authController.ts
import { Request, Response } from 'express';
import { successResponse, errorResponse } from '../utils/response';

export const register = async (req: Request, res: Response) => {
  try {
    // ... 註冊邏輯
    const user = await authService.register(req.body);
    const token = generateToken(user.id);
    
    res.status(201).json(
      successResponse(
        { user, token },
        "註冊成功"
      )
    );
  } catch (error) {
    if (error.code === 'EMAIL_EXISTS') {
      res.status(409).json(
        errorResponse(
          'ALREADY_EXISTS',
          '此 Email 已被註冊',
          { field: 'email' }
        )
      );
    } else {
      res.status(500).json(
        errorResponse(
          'SERVER_ERROR',
          '伺服器發生錯誤',
          process.env.NODE_ENV === 'development' ? error : undefined
        )
      );
    }
  }
};
```

- **GPT-3.5-turbo**: ~$0.0015 / 1K input tokens, ~$0.002 / 1K output tokens
